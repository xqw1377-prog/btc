import os
import sys
import tempfile
import unittest
from decimal import Decimal

sys.path.insert(0, os.path.dirname(os.path.dirname(os.path.abspath(__file__))))

from aurora.capital_controller import Action, CapitalDecision, RiskState
from aurora.execution_truth import (
    DurableExecutor,
    FakeExchange,
    Ledger,
    PositionState,
    Side,
)
from aurora.shadow_adapter import ShadowAdapter


class P0CShadowAdapterTests(unittest.TestCase):
    def setUp(self):
        self.tmp = tempfile.TemporaryDirectory()
        self.db_path = os.path.join(self.tmp.name, "truth.db")
        self.ledger = Ledger(self.db_path)
        self.port = FakeExchange(symbol="BTCUSDT", default_price=Decimal("100000.00"), fee_rate=Decimal("0.0005"))
        self.executor = DurableExecutor(
            "BTCUSDT",
            self.ledger,
            self.port,
            max_retries=2,
            poll_interval_ms=0,
            poll_timeout_ms=0,
        )
        self.adapter = ShadowAdapter(self.executor, Decimal("100000.00"), self.ledger)

    def tearDown(self):
        try:
            self.ledger._conn().close()
            del self.ledger._local.conn
        except Exception:
            pass
        self.tmp.cleanup()

    def test_shadow_open_long(self):
        decision = CapitalDecision(
            action=Action.LONG,
            size=0.1,
            risk_state=RiskState.SAFE,
            stop_loss_pct=0.01,
            take_profit_pct=0.02,
            confidence=0.85,
            reason="test long",
        )
        match = self.adapter.on_decision(decision)
        self.assertTrue(match.agree)
        position = self.ledger.get_position("BTCUSDT")
        self.assertEqual(position.state, PositionState.KNOWN_OPEN)
        self.assertEqual(position.side, Side.LONG)
        self.assertIsNotNone(position.sl_order_id)
        self.assertIsNotNone(position.tp_order_id)

    def test_shadow_open_short(self):
        decision = CapitalDecision(
            action=Action.SHORT,
            size=0.05,
            risk_state=RiskState.SAFE,
            stop_loss_pct=0.02,
            take_profit_pct=0.03,
            confidence=0.80,
            reason="test short",
        )
        match = self.adapter.on_decision(decision)
        self.assertTrue(match.agree)
        position = self.ledger.get_position("BTCUSDT")
        self.assertEqual(position.side, Side.SHORT)

    def test_shadow_flat(self):
        self.adapter.on_decision(
            CapitalDecision(
                action=Action.LONG,
                size=0.1,
                risk_state=RiskState.SAFE,
                stop_loss_pct=0.01,
                take_profit_pct=0.02,
                confidence=0.85,
                reason="test long",
            )
        )
        match = self.adapter.on_decision(
            CapitalDecision(
                action=Action.FLAT,
                size=0,
                risk_state=RiskState.BLOCK,
                stop_loss_pct=0,
                take_profit_pct=0,
                confidence=0,
                reason="risk off",
            )
        )
        self.assertTrue(match.agree)
        self.assertEqual(self.ledger.get_position("BTCUSDT").state, PositionState.KNOWN_FLAT)

    def test_shadow_flat_when_no_position(self):
        match = self.adapter.on_decision(
            CapitalDecision(
                action=Action.FLAT,
                size=0,
                risk_state=RiskState.BLOCK,
                stop_loss_pct=0,
                take_profit_pct=0,
                confidence=0,
                reason="blocked",
            )
        )
        self.assertTrue(match.agree)


if __name__ == "__main__":
    unittest.main()
